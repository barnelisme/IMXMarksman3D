using System.Collections;
using System.Collections.Generic;
using System.Data;
using Unity.VisualScripting.Dependencies.Sqlite;
using UnityEngine;

public class SqlLite : MonoBehaviour
{
    private string connectionString;

    void Start()
    {
        // Set the path to the SQLite database file
        connectionString = "URI=file:" + Application.dataPath + "/example.db";

        // Create a new table called "users" with columns "id" and "name"
        System.Data.IDbConnection dbConnection = (System.Data.IDbConnection)new SQLiteConnection(connectionString);
        dbConnection.Open();
        IDbCommand dbCommand = dbConnection.CreateCommand();
        dbCommand.CommandText = "CREATE TABLE IF NOT EXISTS users (id INTEGER PRIMARY KEY, name TEXT)";
        dbCommand.ExecuteNonQuery();
        dbCommand.Dispose();
        dbConnection.Close();

        // Insert a new row into the "users" table with id=1 and name="John"
        dbConnection = (System.Data.IDbConnection)new SQLiteConnection(connectionString);
        dbConnection.Open();
        dbCommand = dbConnection.CreateCommand();
        dbCommand.CommandText = "INSERT INTO users (id, name) VALUES (1, 'John')";
        dbCommand.ExecuteNonQuery();
        dbCommand.Dispose();
        dbConnection.Close();
    }
}
